// Iterate through the provided text to look for leading modifiers
  enum stateMachine : byte { FIND_MODSTART, FIND_MODIFIER, READ_COLOUR, READ_BACKGROUND };
  size_t plain_text_start = 0;
  int32_t colour = 0;
  stateMachine state = FIND_MODSTART; /** Start by looking for the first backtick */
  auto textLength = strlen(text);
  for (size_t i = 0; i <= textLength; i++) { // loop includes null byte
    auto check = text[i];
    switch (state) {
    case FIND_MODSTART: { // scanning plain text
      // If found backtick or end of line, emit any outstanding text
      if ((check == '`' || check == 0) && i > plain_text_start)
        emitText(text + plain_text_start, i - plain_text_start);

      state = FIND_MODIFIER;
      break;
    }

    case FIND_MODIFIER: { // inside a backtick section
      switch (check) {
      case '`':
        // switch back to plain text
        plain_text_start = i + 1;
        state = FIND_MODSTART;
        break;

      case '#': // foreground
        colour = 0;
        state = READ_COLOUR;
        break;

      case '/':
        colour = 0; // background
        state = READ_BACKGROUND;
        break;

      case '_':
        emitSetUnderline(true);
        break;
      // TODO some others here
      default:
        break; // ignore invalid modifier chars, including spaces
      }
      break;
    }

    case READ_COLOUR:
    case READ_BACKGROUND: { // both states similar until we hit the end, reading hex
      if (check >= '0' && check <= '9') {
        colour = (colour << 4) | check - '0';
        break;
      }
      if (check >= 'a' && check <= 'f') {
        colour = (colour << 4) | check - 'a';
        break;
      }
      if (check >= 'A' && check <= 'F') {
        colour = (colour << 4) | check - 'A';
        break;
      }

           // any non hex char means colour value is finished so tell display
      if (state == READ_COLOUR)
        emitSetColour(colour);
      else
        emitSetBackground(colour);

      // back up the loop so we can re-examine the current char
      i--;
      state = FIND_MODIFIER;
      break;
    }
    }
    // end of loop... all output has been sent to an emit function.
  }
}
